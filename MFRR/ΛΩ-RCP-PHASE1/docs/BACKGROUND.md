```python
import numpy as np
import networkx as nx

def build_srrg_graph(seed, N, params=None):
    rng = np.random.default_rng(seed)
    p1 = 4.0 / N
    p2 = 8.0 / N
    G = nx.Graph()
    G.add_nodes_from(range(N))
    for i in range(N):
        for j in range(i+1, N):
            if rng.random() < (p1 if j-i < 20 else p2):
                G.add_edge(i, j, w=1.0)
    return G

def fisher_metric_proxy(G):
    deg = np.array([G.degree(n) for n in G.nodes()], dtype=float)
    v = np.maximum(deg, 1.0)
    return v

def scalar_curvature_proxy(I):
    inv = 1.0 / I
    R = inv - np.mean(inv)
    return R

def omega_complexity(R, I):
    return float(np.sum(np.abs(R) * np.sqrt(I)))
```

```python
import numpy as np

def random_walk_return_probability(G, t, rng):
    N = G.number_of_nodes()
    if N == 0:
        return 0.0
    walkers = N
    nodes = list(G.nodes())
    pos = rng.integers(0, N, size=walkers)
    start = pos.copy()
    for _ in range(max(1, int(t))):
        for k in range(walkers):
            nbrs = list(G.neighbors(int(pos[k])))
            if len(nbrs) == 0:
                continue
            idx = rng.integers(0, len(nbrs))
            pos[k] = nbrs[idx]
    ret = np.mean(pos == start)
    return float(ret)

def spectral_dimension(G, t_grid, seed):
    rng = np.random.default_rng(seed)
    P0 = []
    for t in t_grid:
        P0.append(random_walk_return_probability(G, max(1, int(round(t))), rng))
    x = np.log(np.maximum(t_grid, 1e-9))
    y = np.log(np.maximum(P0, 1e-12))
    slope = np.polyfit(x, y, 1)[0]
    ds = -2.0 * slope
    return float(ds)
```

```python
# src/rcp/run_rg.py
import numpy as np
import pandas as pd
from .util import ensure_dirs, save_json, set_seed

def srrg_step(g, step):
    m2, lam = g
    dm2 = -0.1*m2 + 0.05*lam
    dlam = -0.02*lam + 0.001*lam*lam
    return np.array([m2 + step*dm2, lam + step*dlam])

def wilson_block_step(g, step, L):
    m2, lam = g
    dm2 = -0.1*m2 + 0.05*lam
    dlam = -0.02*lam + 0.001*lam*lam
    return np.array([m2 + step*dm2, lam + step*dlam])

def estimate_beta_traj(g0, steps, step, method):
    g = np.array(g0, dtype=float)
    traj = [g.copy()]
    betas = []
    for _ in range(steps):
        g_next = method(g, step)
        beta = (g_next - g) / step
        betas.append(beta.copy())
        g = g_next
        traj.append(g.copy())
    return np.array(traj), np.array(betas)

def main():
    ensure_dirs()
    cfg = load_cfg()
    seeds = cfg["seeds"]
    rec = []
    for s in seeds:
        set_seed(s)
        g0 = np.array([1.0, 0.2])
        traj_s, beta_s = estimate_beta_traj(g0, cfg["rg"]["steps"], cfg["rg"]["step_size"], lambda g, st: srrg_step(g, st))
        traj_w, beta_w = estimate_beta_traj(g0, cfg["rg"]["steps"], cfg["rg"]["step_size"], lambda g, st: wilson_block_step(g, st, cfg["rg"]["lattice_size"]))
        err = np.linalg.norm(beta_s - beta_w, axis=1) / (np.linalg.norm(beta_w, axis=1) + 1e-9)
        mean_err = float(np.mean(err))
        rec.append((s, mean_err))
    df = pd.DataFrame(rec, columns=["seed","mean_rel_beta_err"])
    df.to_csv("results/rg_records.csv", index=False)
    pass_flag = float(df["mean_rel_beta_err"].mean()) <= cfg["rg"]["tol_beta_rel"]
    save_json({
        "mean_rel_beta_err": float(df["mean_rel_beta_err"].mean()),
        "tol": cfg["rg"]["tol_beta_rel"],
        "status": "PASS" if pass_flag else "FAIL"
    }, "results/rg_summary.json")
```

```python
# src/rcp/run_pc.py
import numpy as np
import pandas as pd
from .util import ensure_dirs, save_json, set_seed, Lambda

def sample_family(theta, N, rng):
    x = rng.standard_normal(N) + theta
    return x

def fisher_metric_empirical(X, theta):
    var = np.var(X) + 1e-9
    I = np.array([1.0/var])
    return I

def curvature_scalar_empirical(I):
    R = 1.0 / (I + 1e-9)
    return R

def gen_drain_metrics(X):
    gen = float(np.maximum(np.mean(np.abs(X)), 1e-9))
    drain = float(np.maximum(np.std(X), 1e-9))
    return gen, drain

def main():
    ensure_dirs()
    cfg = load_cfg()
    rows = []
    for s in cfg["seeds"]:
        set_seed(s)
        rng = np.random.default_rng(s+11)
        for th in cfg["pc"]["theta_grid"]:
            X = sample_family(th, cfg["pc"]["N_samples"], rng)
            I = fisher_metric_empirical(X, th)
            R = curvature_scalar_empirical(I)
            gen, drain = gen_drain_metrics(X)
            rows.append((s, th, float(R), float(gen/drain)))
    df = pd.DataFrame(rows, columns=["seed","theta","R_scalar","profit"])
    df.to_csv("results/pc_records.csv", index=False)
    x = df["R_scalar"].values
    y = np.log(df["profit"].values)
    a, b = np.polyfit(x, y, 1)
    lambda_expected = cfg["lemma1"]["lambda_expected"]
    pass_flag = abs(a - lambda_expected)/lambda_expected <= cfg["pc"]["tol_lambda_rel"]
    save_json({
        "slope": float(a),
        "intercept": float(b),
        "lambda_expected": lambda_expected,
        "status": "PASS" if pass_flag else "FAIL"
    }, "results/pc_summary.json")

def load_cfg():
    import yaml
    with open("cfg/config.yaml","r") as f:
        return yaml.safe_load(f)

if __name__ == "__main__":
    main()
```
